Bu sorğuların işləməsi üçün cədvəllərindəki sütun adlarının (məsələn: UnitPrice və ya Price) eyni olduğundan əmin olmalısan.
Məlumat bazanda cədvəl adları və ya əlaqələr (foreign keys) fərqlidirmi, yoxsa bu struktur sənə tam uyğun gəldi?




cedveller beledi CREATE TABLE Customers (
    CustomerID INT IDENTITY(1,1) PRIMARY KEY,
    CustomerName NVARCHAR(100) NOT NULL,
    City NVARCHAR(50),
    Gender CHAR(1),
    BirthDate DATE
);

CREATE TABLE Products (
    ProductID INT IDENTITY(1,1) PRIMARY KEY,
    ProductName NVARCHAR(100) NOT NULL,
    Category NVARCHAR(50),
    Price DECIMAL(10,2) NOT NULL
);

CREATE TABLE Orders (
    OrderID INT IDENTITY(1,1) PRIMARY KEY,
    CustomerID INT,
    OrderDate DATE NOT NULL,
    FOREIGN KEY (CustomerID) REFERENCES Customers(CustomerID)
);

CREATE TABLE OrderDetails (
    OrderDetailID INT IDENTITY(1,1) PRIMARY KEY,
    OrderID INT,
    ProductID INT,
    Quantity INT NOT NULL,
    FOREIGN KEY (OrderID) REFERENCES Orders(OrderID),
    FOREIGN KEY (ProductID) REFERENCES Products(ProductID)
);
Verdiyin cədvəl strukturuna (sütun adları: Price, Quantity və s.) tam uyğunlaşdırılmış SQL sorğuları aşağıdadır. Subquery (alt sorğu) tələblərini hər bir bənddə nəzərə almışam.

-- 1) Hər sifariş üçün: CustomerName, OrderDate, Ümumi məbləğ
SELECT 
    c.CustomerName,
    o.OrderDate,
    SUM(od.Quantity * p.Price) AS TotalAmount 
FROM Order o
JOIN Customers c on c.CustomerID = o.CustomerID
JOIN OrderDetails od on od.OrderID=o.OrderID
JOIN Products p ON od.ProductID = p.ProductID 
GROUP BY c.CustomerName,o.OrderDate,;

-- 2) Heç sifarişi olmayan müştərilər
SELECT c.CustomerID, c.CustomerName,c.City
FROM Customers
LEFT JOIN Orders o on o.CustomerID=c.CustomerID
WHERE o.CustomerID IS NULL;

-- 3) Hər məhsul üçün: ProductName, Category, Neçə fərqli sifarişdə olub
SELECT 
    p.ProductName, 
    p.Category,
    COUNT(DISTINCT od.OrderID) AS OrderCount
FROM Products p
LEFT JOIN OrderDetails od on od.ProductID = p.ProductID
GROUP BY p.ProductName,p.Category) ;

-- 4) Şəhər üzrə: Müştəri sayı
SELECT City, COUNT(CustomerID) AS CustomerCount
FROM Customers
GROUP BY City;

-- 5) Hər müştəri üzrə: Ümumi sifariş sayı, Ümumi alış məbləği
SELECT 
    CustomerName,
    COUNT(o.OrderID) TotalOrders,
    SUM(od.Quantity * p.Price) TotalSpent
FROM Customers c
LEFT JOIN ORDERS o ON o.CustomerID = c.CustomerID
LEFT JOIN OrderDetails od ON od.OrderID = o.OrderID
LEFT JOIN Products p ON od.ProductID = p.ProductID
GROUP BY P.CustomerName;

-- 6) Kateqoriya üzrə: Satılan məhsul sayı (Quantity), Ümumi satış məbləği
SELECT 
    p.Category,
    SUM(od.Quantity) AS TotalQuantity,
    SUM(od.Quantity * p.Price) AS TotalRevenue
FROM OrderDetails od
JOIN Products p ON od.ProductID = p.ProductID
GROUP BY p.Category;

-- 7) Heç satılmayan məhsullar (Subquery istifadə et)
SELECT p.ProductID,p.ProductName
FROM Products p
LEFT JOIN OrderDetails od on od.ProductID=p.ProductID

-- 8) Ən çox sifariş verən müştəri (Subquery ilə)

    C.CustomerName, 
    COUNT(O.OrderID) AS SifarisSayi

JOIN Orders O ON C.CustomerID = O.CustomerID
GROUP BY C.CustomerID, C.CustomerName

-- 9) Ən çox gəlir gətirən məhsul
SELECT TOP 1 ProductName
FROM Products p
JOIN OrderDetails od ON p.ProductID = od.ProductID

-- 10) Hər ay üzrə ümumi satış məbləği
SELECT 
    FORMAT(o.OrderDate, 'yyyy-MM') AS SaleMonth,
    SUM(od.Quantity * p.Price) AS MonthlyTotal

JOIN OrderDetails od ON o.OrderID = od.OrderID
JOIN Products p ON od.ProductID = p.ProductID

-- 11) Ən çox məhsul (Quantity) alan müştəri

WHERE CustomerID = (
    SELECT TOP 1 o.CustomerID
    FROM Orders o
    JOIN OrderDetails od ON o.OrderID = od.OrderID

-- 12) Orta sifariş məbləği
SELECT AVG(OrderTotals.Total) AS AverageOrderValue
FROM (
    select(
    od.OrderID, 
    SUM(od.Quantity * p.Price) AS Total
FROM OrderDetails od
JOIN Products p ON od.ProductID = p.ProductID
GROUP BY od.OrderID
) AS OrderTotals;

-- 14) 1000-dən çox alış edən müştərilər

FROM Customers c

JOIN OrderDetails od ON o.OrderID = od.OrderID
